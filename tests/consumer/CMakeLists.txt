cmake_minimum_required(VERSION 3.20)
project(CertPolySolveConsumer LANGUAGES C CXX)
find_package(CertPolySolve CONFIG REQUIRED)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE CertPolySolve::certpolysolve)
add_executable(consumer_c solve.c)
target_link_libraries(consumer_c PRIVATE CertPolySolve::certpolysolve)
enable_testing()
add_test(NAME installed_cpp COMMAND consumer)
add_test(NAME installed_c COMMAND consumer_c)
